if(WIN32)
	set(launcher_curl_tls "CURL_USE_SCHANNEL ON" "CURL_USE_MBEDTLS OFF")
else()
	set(launcher_curl_tls "CURL_USE_SCHANNEL OFF" "CURL_USE_MBEDTLS ON")
	# cURL's FindMbedTLS accepts existing CMake targets as its libraries.
	set(MBEDTLS_INCLUDE_DIR "${LAUNCHER_MBEDTLS_INCLUDE}")
	set(MBEDTLS_LIBRARY mbedtls)
	set(MBEDX509_LIBRARY mbedx509)
	set(MBEDCRYPTO_LIBRARY mbedcrypto)
	# The freshly added target cannot participate in try_compile until built.
	set(HAVE_MBEDTLS_DES_CRYPT_ECB 1)
endif()
CPMAddPackage(
	NAME CURL
	VERSION 8.23.0
	URL https://curl.se/download/curl-8.23.0.tar.xz
	URL_HASH SHA256=8dadd7d2536a314f982ad05a651b713f426bf7d02cc2c76cffb9796f2218e6c5
	EXCLUDE_FROM_ALL YES
	OPTIONS
		"BUILD_SHARED_LIBS OFF"
		"BUILD_STATIC_LIBS ON"
		"BUILD_CURL_EXE OFF"
		"BUILD_TESTING OFF"
		"CMAKE_POSITION_INDEPENDENT_CODE ON"
		"CURL_DISABLE_INSTALL ON"
		"CURL_USE_OPENSSL OFF"
		"CURL_USE_PKGCONFIG OFF"
		"CURL_USE_CMAKECONFIG OFF"
		"CURL_USE_LIBPSL OFF"
		"CURL_USE_LIBSSH2 OFF"
		"CURL_USE_LIBSSH OFF"
		"CURL_USE_GSSAPI OFF"
		"CURL_ZLIB OFF"
		"CURL_BROTLI OFF"
		"CURL_ZSTD OFF"
		"USE_NGHTTP2 OFF"
		"USE_LIBIDN2 OFF"
		"CURL_DISABLE_LDAP ON"
		"CURL_DISABLE_LDAPS ON"
		${launcher_curl_tls}
)
